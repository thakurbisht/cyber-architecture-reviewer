# add_ground_truth.ps1
# Adds ground truth annotations to sample files

param(
    [string]$ProjectRoot = (Get-Location)
)

Write-Host "Adding ground truth annotations to samples..."

# Sample 1: campus LAN
$file1 = Join-Path $ProjectRoot "samples\sample-campus-lan-lld.md"
if (Test-Path $file1) {
    $content = Get-Content $file1 -Raw
    if ($content -notmatch "EXPECTED CRITICAL") {
        $insert = @"
<!-- EXPECTED CRITICAL: -->
- Section "5. Distribution and Core": single point of failure uplink device
- Section "7. WAN Edge and Internet": single circuit uplink failure
- Section "3. Access Layer Design": uplinks terminate same upstream device
- Section "10. Management Plane": SNMPv2c community cleartext
- Section "10. Management Plane": community string public private guessable
- Section "8. Firewall and Segmentation": permit-any-any firewall ACL rule
- Section "10. Management Plane": admin access without MFA authentication

<!-- EXPECTED HIGH: -->
- Section "10. Management Plane": telnet TFTP cleartext device management
- Section "10. Management Plane": local accounts shared credentials enable password
- Section "6. Routing Design": OSPF neighbour authentication dynamic routing
- Section "4. VLAN and Addressing Design": flat VLAN single broadcast domain
- Section "7. WAN Edge and Internet": WAN traffic untrusted transport encryption
- Section "9. Wireless": wireless pre-shared key deprecated security mode
- Section "9. Wireless": passphrase shared service desk generic account
- Section "Whole document": central log aggregation SIEM forwarding
- Section "10. Management Plane": backups immutability offline tested restore
- Section "11. Availability and Convergence": backups immutability offline tested

<!-- EXPECTED MEDIUM: -->
- Section "Whole document": threat model attack surface analysis STRIDE

"@
        $lines = @(Get-Content $file1)
        $insertIdx = 0
        for ($i = 0; $i -lt $lines.Count; $i++) {
            if ($lines[$i] -match "Site tier:") {
                $insertIdx = $i + 1
                break
            }
        }
        
        if ($insertIdx -gt 0) {
            $newLines = $lines[0..$insertIdx] + $insert.Split("`n") + $lines[($insertIdx+1)..($lines.Count-1)]
            $newLines | Out-File -FilePath $file1 -Encoding UTF8
            Write-Host "[OK] Added ground truth to sample-campus-lan-lld.md"
        }
    } else {
        Write-Host "[SKIP] Ground truth already in sample-campus-lan-lld.md"
    }
}
